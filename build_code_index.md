# build\_code\_index.py

Save as `scripts/build_code_index.py`. Run with `uv run scripts/build_code_index.py`. It reads every Python and TypeScript file and writes `CODE_INDEX.md` at the project root.

Every function needs a docstring (Python) or a comment block above it (TypeScript) in this form: what it does and why it exists, plus a version in the form `v1_0`. Example: `"""Pay a claim for the given amount. Exists so payment is explicit. v1_1"""`.

```python
"""Build CODE_INDEX.md from the docstrings in the project. Exists so agents can see what code exists. v1_0"""

import ast
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
SKIP_DIRS = {".venv", "node_modules", ".git", "dist", "build", "__pycache__", "scripts"}
VERSION = re.compile(r"\bv(\d+_\d+)\b")
HEADER = (
    "# Code index\n\n"
    "Generated by `scripts/build_code_index.py`. Never edit by hand. "
    "Each entry starts with its versioned id, which FEATURES.md links to.\n"
)


def split_doc(doc):
    """Split a docstring into its text and its version. Exists so ids carry the version. v1_0"""
    doc = (doc or "").strip()
    match = VERSION.search(doc)
    version = match.group(1) if match else "0_0"
    text = VERSION.sub("", doc).strip()
    return " ".join(text.split()), version


def python_entry(module, owner, node):
    """Make one index line for a Python function or method. Exists to keep the line format in one place. v1_0"""
    text, version = split_doc(ast.get_docstring(node))
    args = ", ".join(a.arg for a in node.args.args if a.arg not in ("self", "cls"))
    name = f"{owner}.{node.name}" if owner else node.name
    return f"- `{module}.{name}_v{version}` `{node.name}({args})`: {text or 'No docstring.'}"


def index_python(path):
    """Index one Python file into a summary, classes and functions. Exists so Python gets the shared format. v1_0"""
    tree = ast.parse(path.read_text(encoding="utf-8"))
    module = path.stem
    summary, _ = split_doc(ast.get_docstring(tree))
    lines = [summary, ""] if summary else []
    functions = []
    for node in tree.body:
        if isinstance(node, ast.ClassDef):
            lines.append(f"### class {node.name}\n")
            for item in node.body:
                if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    lines.append(python_entry(module, node.name, item))
            lines.append("")
        elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            functions.append(python_entry(module, "", node))
    if functions:
        lines += ["### Functions\n", *functions, ""]
    return lines


TS_FUNCTION = re.compile(
    r"/\*\*(?P<doc>.*?)\*/\s*export\s+(?:async\s+)?function\s+(?P<name>\w+)\s*\((?P<args>[^)]*)\)",
    re.S,
)


def index_typescript(path):
    """Index one TypeScript file from its comment blocks. Exists so TypeScript gets the shared format. v1_0"""
    module = path.stem
    entries = []
    for match in TS_FUNCTION.finditer(path.read_text(encoding="utf-8")):
        doc = re.sub(r"^\s*\*\s?", "", match["doc"], flags=re.M)
        text, version = split_doc(doc)
        args = ", ".join(a.split(":")[0].strip() for a in match["args"].split(",") if a.strip())
        entries.append(f"- `{module}.{match['name']}_v{version}` `{match['name']}({args})`: {text}")
    return ["### Functions\n", *entries, ""] if entries else []


READERS = {".py": index_python, ".ts": index_typescript, ".tsx": index_typescript}


def source_files():
    """List the code files to index, in a stable order. Exists so the index only changes when code does. v1_0"""
    for path in sorted(ROOT.rglob("*")):
        if path.suffix in READERS and not SKIP_DIRS & set(path.relative_to(ROOT).parts):
            yield path


def main():
    """Write CODE_INDEX.md for the whole project. Exists as the one command that rebuilds the index. v1_0"""
    out = [HEADER]
    for path in source_files():
        body = READERS[path.suffix](path)
        if body:
            out += [f"## {path.relative_to(ROOT).as_posix()}\n", *body]
    (ROOT / "CODE_INDEX.md").write_text("\n".join(out), encoding="utf-8")
    print(f"CODE_INDEX.md written for {len(out) - 1} lines.")


if __name__ == "__main__":
    main()
```

## How it behaves

- Ids are the file name, the function name and the version, such as `claims.pay_claim_v1_1`. A methods id includes its class.
- A function with no version in its docstring gets `v0_0`, which makes missing versions easy to spot.
- The `scripts/` folder is skipped, so helper scripts never clutter the index.
- Output order is sorted, so the file only changes when the code does.
- To support another language, add a reader function and register it in `READERS`.
